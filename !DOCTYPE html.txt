    async function saveToCloud() {
      const docId = document.getElementById('editDocId').value;
      const sec = document.getElementById('uploadSec').value;
      const cls = document.getElementById('uploadClass').value;
      const updatedOn = new Date().toLocaleDateString('te-IN');

      let itemData = { sec, cls, updatedOn, timestamp: Date.now() };

      if (sec === 'quiz') {
        itemData.type = 'online_quiz';
        itemData.qText = document.getElementById('quizQText').value;
        itemData.optA = document.getElementById('quizOptA').value;
        itemData.optB = document.getElementById('quizOptB').value;
        itemData.optC = document.getElementById('quizOptC').value;
        itemData.optD = document.getElementById('quizOptD').value;
        itemData.correct = document.getElementById('quizCorrectOpt').value;
        if (!itemData.qText) { alert("ప్రశ్న నమోదు చేయండి!"); return; }
      } else {
        itemData.title = document.getElementById('uploadTitle').value;
        itemData.type = document.getElementById('fileType').value;
        itemData.link = document.getElementById('uploadUrl').value;
        if (!itemData.title || !itemData.link) { alert("శీర్షిక & లింక్ తప్పనిసరి!"); return; }
      }

      try {
        if (isCloudReady && db) {
          if (docId) {
            await db.collection("materials").doc(docId).update(itemData);
            alert("నవీకరించబడింది (Updated)!");
          } else {
            await db.collection("materials").add(itemData);
            alert("క్లౌడ్‌లో సేవ్ చేయబడింది! అందరికీ కనిపిస్తుంది.");
          }
        }
      } catch(err) { alert("Error: " + err.message); }

      document.getElementById('editDocId').value = '';
      document.getElementById('uploadTitle').value = '';
      document.getElementById('uploadUrl').value = '';
      document.getElementById('quizQText').value = '';
      closeAdmin();
    }

    }
